# K-Buddy API — CPU 기본 설치 (Docker · CI · macOS)
#
#     pip install -r requirements.txt
#
# 로컬 생성(`LLM_BACKEND=local`)을 **실측할 목적이라면 이 파일이 아니다.**
# ADR-005 가 "CUDA 실측으로만 판정한다"고 못박았으므로 개발·판정에는
# `requirements-gpu.txt` 를 쓴다. 이 파일로 깔면 torch 가 CPU 로 돌아
# 4B 를 fp32 로 굴리게 되며 사실상 멈춘 것처럼 보인다.
#
# 이 파일이 맞는 경우:
#   - Docker 이미지 빌드 (GPU 런타임을 붙이지 않는 현재 Dockerfile)
#   - CI / 테스트 (로컬 생성 경로를 타지 않는다)
#   - macOS 개발 — ADR-005 로 MPS 를 걷어낸 뒤로 Apple Silicon 에는
#     GPU 경로가 없다. API·검색·프런트 작업에는 영향이 없다.

-r requirements-base.txt

# ── 장치 의존 ────────────────────────────────────────────────────────
# PyPI 기본 휠. Linux 는 nvidia-cu13 을 자동 동반해 GPU 가 있으면 쓰지만,
# Windows 휠(116MB)은 **CPU 전용**이다 — requirements-gpu.txt 참조.
torch~=2.13

# 명시적으로 고정한다. 이걸 적지 않으면 chromadb·fastembed 가 전이 의존으로
# 끌어와, GPU 환경에서 `onnxruntime-gpu` 를 깐 뒤에도 조용히 되살아난다.
onnxruntime~=1.29
